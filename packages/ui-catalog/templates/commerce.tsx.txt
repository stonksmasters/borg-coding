import { useState } from "react";
import { Modal, Navigation, Notice } from "./primitives";

const products = [{ id: "01", name: "Everyday tote", category: "Carry", price: 48 }, { id: "02", name: "Desk vessel", category: "Home", price: 32 }, { id: "03", name: "Weekender", category: "Carry", price: 96 }];
export function CommercePage() {
  const [filter, setFilter] = useState("All"); const [cart, setCart] = useState<Record<string, number>>({}); const [open, setOpen] = useState(false); const [done, setDone] = useState(false);
  const [detail, setDetail] = useState<string | null>(null);
  const selected = products.find((item) => item.id === detail);
  const count = Object.values(cart).reduce((sum, quantity) => sum + quantity, 0);
  return <div className="borg-page"><Navigation brand="Considered goods" links={[{ href: "#collection", label: "Collection" }]} /><main><section className="borg-hero"><div><p className="borg-eyebrow">A smaller, better collection</p><h1>Useful things.<br />Beautifully considered.</h1><p>Illustrative products for a working local storefront demonstration.</p><a className="borg-button" href="#collection">Browse collection</a></div><div className="borg-art" aria-hidden="true"><span /><span /></div></section><section id="collection" className="borg-section"><div className="borg-section-heading"><h2>The collection</h2><button type="button" onClick={() => { setDone(false); setOpen(true); }}>Cart ({count})</button></div><label>Category <select value={filter} onChange={(event) => setFilter(event.target.value)}><option>All</option><option>Carry</option><option>Home</option></select></label><div className="borg-grid">{products.filter((item) => filter === "All" || item.category === filter).map((item, index) => <article key={item.id}><div className={`borg-swatch borg-swatch-${index}`} /><p className="borg-eyebrow">{item.category}</p><h3><button type="button" onClick={() => setDetail(item.id)}>{item.name}</button></h3><p>${item.price}</p><button type="button" className="borg-button" onClick={() => { setCart({ ...cart, [item.id]: (cart[item.id] ?? 0) + 1 }); setDone(false); }}>Add {item.name} to cart</button></article>)}</div><p role="status">{count} items in cart</p></section></main>
    <Modal open={Boolean(selected)} onClose={() => setDetail(null)} title={selected?.name ?? "Product"}><p>An illustrative product with a clean, functional design.</p><p>${selected?.price}</p></Modal>
    <Modal open={open} onClose={() => setOpen(false)} title="Your cart" drawer>{count === 0 ? <p>Your cart is empty.</p> : <>{products.filter((item) => cart[item.id]).map((item) => <div className="borg-cart-row" key={item.id}><span>{item.name}</span><label>Quantity for {item.name}<input type="number" min={1} max={99} value={cart[item.id]} onChange={(event) => setCart({ ...cart, [item.id]: Math.max(1, Math.min(99, Number(event.target.value) || 1)) })} /></label><button type="button" onClick={() => { const next = { ...cart }; delete next[item.id]; setCart(next); }}>Remove {item.name}</button></div>)}<p>Total: ${products.reduce((sum, item) => sum + item.price * (cart[item.id] ?? 0), 0)}</p><button type="button" className="borg-button" onClick={() => { setCart({}); setDone(true); }}>Complete demo checkout</button></>}{done && <Notice>Demo checkout complete. No payment was taken and no order was placed.</Notice>}</Modal></div>;
}
