import { useState } from "react";
import { Modal, Notice } from "./primitives";

const seed = [{ id: "1", name: "Website refresh", status: "In progress", owner: "Alex" }, { id: "2", name: "Spring collection", status: "Planned", owner: "Morgan" }, { id: "3", name: "Customer guide", status: "Complete", owner: "Sam" }];
export function WorkspacePage() {
  const [records, setRecords] = useState(seed); const [query, setQuery] = useState(""); const [descending, setDescending] = useState(false);
  const [state, setState] = useState("ready"); const [selected, setSelected] = useState<string | null>(null); const [notice, setNotice] = useState("");
  const rows = records.filter((item) => item.name.toLowerCase().includes(query.toLowerCase())).sort((a, b) => a.name.localeCompare(b.name) * (descending ? -1 : 1));
  return <div className="borg-workspace"><aside><a className="borg-brand" href="#projects">Studio desk</a><nav aria-label="Workspace"><a href="#projects">Projects</a><a href="#settings">Demo settings</a></nav><p>Local demonstration data</p></aside><main><header className="borg-section-heading"><div><p className="borg-eyebrow">Your workspace</p><h1>Projects</h1></div><button type="button" onClick={() => { setRecords(seed); setQuery(""); setState("ready"); setNotice("Demo data reset."); }}>Reset demo</button></header><section id="projects"><label className="borg-field">Find a project<input type="search" value={query} onChange={(event) => setQuery(event.target.value)} /></label>{state === "loading" ? <Notice>Loading demonstration projects…</Notice> : state === "error" ? <><Notice error>Unable to load demonstration data.</Notice><button type="button" onClick={() => setState("ready")}>Try again</button></> : state === "empty" || !rows.length ? <Notice>No projects match. Adjust your search or reset the demo.</Notice> : <div className="borg-table-wrap"><table><caption>Demonstration projects</caption><thead><tr><th scope="col" aria-sort={descending ? "descending" : "ascending"}><button type="button" onClick={() => setDescending(!descending)}>Project {descending ? "↓" : "↑"}</button></th><th scope="col">Status</th><th scope="col">Owner</th><th scope="col">Action</th></tr></thead><tbody>{rows.map((item) => <tr key={item.id}><th scope="row">{item.name}</th><td>{item.status}</td><td>{item.owner}</td><td><button type="button" onClick={() => setSelected(item.id)}>Edit {item.name}</button></td></tr>)}</tbody></table></div>}{notice && <Notice>{notice}</Notice>}</section><section id="settings" className="borg-section"><h2>Demo settings</h2><label>Display state <select value={state} onChange={(event) => setState(event.target.value)}>{["ready", "loading", "empty", "error"].map((item) => <option key={item}>{item}</option>)}</select></label></section></main><Modal open={selected !== null} onClose={() => setSelected(null)} title="Edit project"><form key={selected} onSubmit={(event) => { event.preventDefault(); const data = new FormData(event.currentTarget); const name = String(data.get("name") ?? "").trim(); if (!name) return; setRecords(records.map((item) => item.id === selected ? { ...item, name } : item)); setSelected(null); setNotice("Project updated locally."); }}><label className="borg-field">Project name<input name="name" required pattern=".*\S.*" defaultValue={records.find((item) => item.id === selected)?.name} /></label><button type="submit" className="borg-button">Save changes</button></form></Modal></div>;
}
